**Задача 1.**
select case when
reg_date BETWEEN '2022-11-01' and '2022-12-31' then 'Когорта 11-12.22' else 'Остальные когорты' end cohorts
                , avg(end_session - start_session) avg_sesion
from skygame.users u
               left join skygame.game_sessions s
                 
on u.id_user = s.id_user
where end_session - start_session > interval '5
minute'
group by cohorts


**задача 2.**
with k_factor_data as
(select sum(r.ref_reg) as cnt_ref_reg   
,count(distinct u.id_user) as cnt_users
from skygame.users u
left join skygame.referral r
on u.id_user = r.id_user),
cohort as
(select cnt_ref_ref_reg          
,cnt_users      
,cnt_ref_reg*1.0 /cnt_users as k_factor
from k_factor_data),
historical_avg_cohort as
(select count(distinct id_user) * 1.0 /
count(distinct date_trunc('month', reg_date)) as avg_cohort_size
from skygame.users)
select avg_cohort_size * k_factor as
expected_cohort_size
from cohort
cross join historical_avg_cohort
